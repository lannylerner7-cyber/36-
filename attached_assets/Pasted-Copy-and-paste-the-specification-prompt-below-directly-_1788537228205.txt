3. MICR Engine Implementation Logic
Implement a backend utility function that formats the standard ANSI X3.2 E-13B MICR line.
// Glyphs: 'A' or '⑆' = Transit Symbol | 'C' or '⑈' = On-Us Symbol
export function generateMICRLine(routing: string, account: string, checkNum: string): string {
  const cleanRouting = routing.replace(/\D/g, '');
  const cleanAccount = account.replace(/\D/g, '');
  const cleanCheck = checkNum.replace(/\D/g, '');

  if (cleanRouting.length !== 9) return '';

  // Standard E-13B String Sequence: ⑆[Routing]⑆ [Account]⑈ [CheckNumber]⑈
  return `A${cleanRouting}A ${cleanAccount}C ${cleanCheck}C`;
}
